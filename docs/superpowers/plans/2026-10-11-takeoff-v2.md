> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make materially look and work like a desktop takeoff tool in DrAIwall's design language. It has two pages:
- **Projects:** upload a set, list projects, and add alternates under each project.
- **Takeoff:** a sheet dropdown, the scale beside Undo, and conditions on the left. An auto-scan button builds the library, and each condition carries its materials and their product data. Orders, EWOs, Tracking and Review open as popups.

**Architecture:** Same stack: FastAPI, server-rendered Jinja and vanilla JS, MongoDB.
- **Pages:** only `/` and `/jobs/{id}/takeoff`. Every other screen becomes a `<dialog>` popup on the takeoff page, filled with server-rendered partials and driven by the JSON APIs that already exist.
- **New data:**
  - an `alternates` collection (a project has a base bid plus named alternates)
  - `alternate_id` on measurements and orders
  - a `materials[]` array on conditions, derived from the parsed type and its callout text, with no extra model call
- **New code:**
  - a `projects/` package for the project and alternate APIs
  - per-material product search in `procure/`, cached in `product_data` and spec-checked by Jev

**Tech stack:** Python 3.13, FastAPI, Jinja2, pymongo / mongomock, pdfium, vanilla JS (`ui/static/*.js`), plain CSS with tokens (no Tailwind, no build step).

**Spec:** Ethan's requests of 15:15, listed below as R1–R8, plus `docs/SPEC.md` (§4 contracts, §6 router) and the DrAIwall visual reference (`C:\Users\ethan\dryaiwall\src\app\globals.css`, `src\app\app\layout.tsx`, `src\app\app\_components\nav.tsx`). Executors read all three.

```python
"""Project alternates. Drawings and the conditions library belong to the project; measurements, quantities and
orders belong to one alternate. Every project has a Base bid, made on first use. A measurement or order with no
alternate_id counts as the base bid (everything made before alternates existed)."""
from pymongo.errors import DuplicateKeyError

from app.db import get_db, jsonable, now, oid

BASE_NAME = "Base bid"


def ensure_base(job_id: str, db=None) -> dict:
    db = db if db is not None else get_db()
    base = db.alternates.find_one({"job_id": job_id, "kind": "base"})
    if base:
        return base
    doc = {"job_id": job_id, "name": BASE_NAME, "kind": "base", "description": None, "order": 0, "created_at": now()}
    try:
        doc["_id"] = db.alternates.insert_one(doc).inserted_id
    except DuplicateKeyError:  # two first requests at once: the other one made it
        return db.alternates.find_one({"job_id": job_id, "kind": "base"})
    return doc


def list_alternates(job_id: str) -> list[dict]:
    db = get_db()
    ensure_base(job_id, db)
    return jsonable(list(db.alternates.find({"job_id": job_id}).sort([("order", 1), ("created_at", 1)])))


def _clean(name: str) -> str:
    name = " ".join((name or "").split())
    if not name:
        raise ValueError("An alternate needs a name.")
    return name[:80]


def create_alternate(job_id: str, name: str, description: str | None = None) -> str:
    db = get_db()
    ensure_base(job_id, db)
    doc = {"job_id": job_id, "name": _clean(name), "kind": "alternate", "description": description,
           "order": db.alternates.count_documents({"job_id": job_id}), "created_at": now()}
    try:
        return str(db.alternates.insert_one(doc).inserted_id)
    except DuplicateKeyError:
        raise ValueError(f"This project already has an alternate called {doc['name']!r}.") from None


def _get(alt_id: str) -> dict:
    a = get_db().alternates.find_one({"_id": oid(alt_id)}) if oid(alt_id) else None
    if a is None:
        raise LookupError("No such alternate.")
    return a


def rename_alternate(alt_id: str, name: str) -> None:
    a = _get(alt_id)
    try:
        get_db().alternates.update_one({"_id": a["_id"]}, {"$set": {"name": _clean(name)}})
    except DuplicateKeyError:
        raise ValueError("Another alternate already has that name.") from None


class InUse(RuntimeError):
    pass


def delete_alternate(alt_id: str) -> None:
    db, a = get_db(), _get(alt_id)
    if a["kind"] == "base":
        raise InUse("The base bid can't be deleted.")
    if db.measurements.count_documents({"alternate_id": alt_id}, limit=1):
        raise InUse("This alternate has measurements. Delete them first.")
    db.alternates.delete_one({"_id": a["_id"]})


def scope(job_id: str, alternate_id: str | None) -> dict:
    """Filter fragment selecting one alternate's measurements or orders. None or the base id = base bid,
    which also owns the docs that predate alternates (no alternate_id)."""
    base = str(ensure_base(job_id)["_id"])
    if not alternate_id or alternate_id == base:
        return {"alternate_id": {"$in": [base, None]}}
    return {"alternate_id": alternate_id}


def project_rows() -> list[dict]:
    """Every project, newest first, with its alternates: what the projects page lists."""
    db = get_db()
    jobs = jsonable(list(db.jobs.find().sort("created_at", -1).limit(200)))
    for j in jobs:
        j["alternates"] = list_alternates(j["id"])
    return jobs
```

```python
"""Projects and their alternates (Base bid, Alternate 1, ...)."""
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel

from projects import alternates as alt

router = APIRouter(prefix="/api", tags=["projects"])


class AlternateIn(BaseModel):
    name: str
    description: str | None = None


@router.get("/projects")
def projects():
    return alt.project_rows()


@router.get("/jobs/{job_id}/alternates")
def list_alternates(job_id: str):
    return alt.list_alternates(job_id)


@router.post("/jobs/{job_id}/alternates")
def create_alternate(job_id: str, body: AlternateIn):
    try:
        return {"id": alt.create_alternate(job_id, body.name, body.description)}
    except ValueError as e:
        raise HTTPException(422, str(e))


@router.patch("/alternates/{alt_id}")
def rename_alternate(alt_id: str, body: AlternateIn):
    try:
        alt.rename_alternate(alt_id, body.name)
    except LookupError as e:
        raise HTTPException(404, str(e))
    except ValueError as e:
        raise HTTPException(422, str(e))
    return {"id": alt_id}


@router.delete("/alternates/{alt_id}")
def delete_alternate(alt_id: str):
    try:
        alt.delete_alternate(alt_id)
    except LookupError as e:
        raise HTTPException(404, str(e))
    except alt.InUse as e:
        raise HTTPException(409, str(e))
    return {"deleted": alt_id}
```

```python
"""A condition's materials: what a wall type is built from, one row per material, so each can be priced and
matched to product data. Built from the parsed condition plus the type's callout text (Block.text). No model call."""
import re
from collections import Counter
from fractions import Fraction

BOARD = {
    "regular": "regular", "type_x": "Type X", "type_c": "Type C", "moisture_resistant": "moisture-resistant",
    "mold_resistant": "mold-resistant", "type_x_mold_resistant": "Type X mold-resistant", "abuse_resistant": "abuse-resistant",
    "impact_resistant": "impact-resistant", "shaftliner": "shaftliner", "cement_board": "cement", "glass_mat_sheathing": "glass-mat",
    "sound_dampening": "sound-dampening",
}
NO_INSULATION = {"", "none", "no", "n/a", "na", "-"}
RC = re.compile(r"(\d{2})\s*GA\.?\s*(?:SINGLE\s+LEG\s+)?RESILIENT\s+CHANNEL(?:\s*@\s*(\d+)\s*\"?\s*O\.?C)?")


def _in(v) -> str:
    whole = int(v)
    frac = Fraction(v - whole).limit_denominator(16)
    if not frac:
        return f'{whole}"'
    return f'{frac.numerator}/{frac.denominator}"' if not whole else f'{whole}-{frac.numerator}/{frac.denominator}"'


def _m(key, role, desc, per, source, **attrs) -> dict:
    """Attributes stay even when null: a null gauge means "left to the specs", which the order math assumes."""
    return {"key": key, "role": role, "desc": desc, "per": per, "source": source, **attrs}


def derive(condition: dict, text: str = "") -> list[dict]:
    t = " ".join((text or "").upper().split())
    studs = condition.get("studs") or {}
    size, gauge, spacing = studs.get("size_in"), studs.get("gauge"), studs.get("spacing_in")
    sides = condition.get("sides") or 2
    out = []
    if size:
        g = f" {gauge} ga" if gauge else ""
        out.append(_m("stud", "stud", f"{_in(size)}{g} metal studs" + (f" @ {_in(spacing)} o.c." if spacing else ""), "lf", "drawing",
                      size_in=size, gauge=gauge, spacing_in=spacing))
        out.append(_m("track", "track", f"{_in(size)}{g} track, top and bottom", "lf", "derived", size_in=size, gauge=gauge))
    for (thick, kind), n in Counter((l["thickness_in"], l["board_type"]) for l in condition.get("layers_each_side") or []).items():
        where = "each side" if sides == 2 else "one side"
        out.append(_m(f"board_{thick:g}_{kind}", "board", f"{_in(thick)} {BOARD.get(kind, kind)} gypsum board, {n} layer{'s' * (n > 1)} {where}",
                      "sf", "drawing", thickness_in=thick, board_type=kind, sides=sides))
    insulation = str(condition.get("insulation") or "").strip()
    if insulation.lower() not in NO_INSULATION or "BATT" in t:
        out.append(_m("insulation", "insulation", insulation or "Acoustical batt insulation, full cavity", "sf", "drawing", size_in=size))
    if rc := RC.search(t):
        one_side = "ONE SIDE" in t[rc.end(): rc.end() + 60]
        out.append(_m("resilient_channel", "resilient_channel", f"{rc.group(1)} ga resilient channel" + (f" @ {rc.group(2)}\" o.c." if rc.group(2) else "")
                      + (", one side" if one_side else ", both sides"), "lf", "drawing",
                      gauge=int(rc.group(1)), spacing_in=float(rc.group(2)) if rc.group(2) else None, sides=1 if one_side else 2))
    if "FURRING CHANNEL" in t:
        out.append(_m("furring", "furring", "Furring channel", "lf", "drawing"))
    if "SEALANT" in t:
        rated = bool(condition.get("fire_rating"))
        out.append(_m("sealant_firerated" if rated else "sealant_acoustical", "sealant",
                      "Fire-rated acoustical sealant, top and bottom of wall" if rated else "Acoustical sealant, top and bottom of wall", "lf", "drawing"))
    if "BRACE" in t:
        out.append(_m("bracing", "bracing", "Diagonal bracing back to structure @ 48\" max", "ea", "drawing"))
    return out
```

```python
"""Product data for a condition's materials: one supplier search per material, cached by query hash so a
repeat search makes no Firecrawl call; attributes read by run_task; each specified attribute checked by Jev."""
import hashlib

import httpx

from app import jev
from app.db import get_db, jsonable, now, oid
from app.router import TaskFailed, run_task
from app.settings import get_settings

FIRECRAWL = "https://api.firecrawl.dev/v1/search"
CHECKED = ("thickness_in", "board_type", "gauge", "size_in")  # material attributes a product must match


class Unavailable(RuntimeError):
    pass


def _key() -> str:
    return get_settings().firecrawl_api_key


def query_for(material: dict) -> str:
    return f"{material['desc'].split(',')[0]} product data sheet"


def _search(query: str) -> list[dict]:
    r = httpx.post(FIRECRAWL, headers={"Authorization": f"Bearer {_key()}"},
                   json={"query": query, "limit": 3, "scrapeOptions": {"formats": ["markdown"]}}, timeout=60)
    r.raise_for_status()
    return r.json().get("data") or []


def _attrs(text: str, job_id: str) -> dict:
    return run_task("extract_product_attrs", text[:12000], "product_attrs", job_id=job_id)["result"]


def _p_mismatch(material: dict, attrs: dict, attr: str, job_id: str) -> float:
    q = f"Does this product fail the specified {attr}: {material.get(attr)}?"
    return jev.noul({"specified": material, "product": attrs}, q, task="jev.spec_check", job_id=job_id)["p_yes"]


def find_for_material(condition_id: str, key: str) -> dict:
    db = get_db()
    cond = db.conditions.find_one({"_id": oid(condition_id)}) if oid(condition_id) else None
    material = next((m for m in (cond or {}).get("materials") or [] if m["key"] == key), None)
    if material is None:
        raise LookupError(f"No material {key!r} on this condition.")
    query = query_for(material)
    qhash = hashlib.sha1(query.lower().encode()).hexdigest()
    doc = db.product_data.find_one({"query_hash": qhash})
    if doc is None:
        if not _key():
            raise Unavailable("Product search needs a Firecrawl key (FIRECRAWL_API_KEY in Infisical).")
        pages = _search(query)
        if not pages:
            raise LookupError(f"No product data found for {query!r}.")
        top = pages[0]
        try:
            attrs = _attrs(f"{top.get('title')}\n{top.get('markdown') or ''}", cond.get("job_id"))
        except TaskFailed as e:
            raise LookupError(f"Couldn't read the product sheet: {e.reason}") from None
        checks = {a: round(_p_mismatch(material, attrs, a, cond.get("job_id")), 3) for a in CHECKED if material.get(a) is not None}
        doc = {"query_hash": qhash, "condition_id": condition_id, "component": key, "url": top.get("url"),
               "title": top.get("title"), "attrs": attrs, "spec_check": checks, "fetched_at": now()}
        doc["_id"] = db.product_data.insert_one(doc).inserted_id
    db.conditions.update_one({"_id": cond["_id"], "materials.key": key}, {"$set": {"materials.$.product_id": str(doc["_id"])}})
    return jsonable(doc)
```

```python
def _partial(request: Request, name: str, **ctx):
    return templates.TemplateResponse(request, name, ctx)


@router.get("/ui/jobs/{job_id}/panel/orders", response_class=HTMLResponse)
def panel_orders(request: Request, job_id: str, alt: str | None = None):
    db = get_db()
    q = {"job_id": job_id, **(scope(job_id, alt) if alt else {})}
    orders = jsonable(list(db.orders.find(q).sort("_id", -1)))
    conds = {c["id"]: c for c in jsonable(list(db.conditions.find({"job_id": job_id})))}
    return _partial(request, "_orders.html", orders=orders, conditions=conds)


@router.get("/ui/orders/{order_id}/panel", response_class=HTMLResponse)
def panel_order(request: Request, order_id: str):
    db = get_db()
    order = db.orders.find_one({"_id": oid(order_id)})
    if not order:
        raise HTTPException(404, "No such order.")
    order = jsonable(order)
    over = [x.get("item") if isinstance(x, dict) else x for x in (order.get("guard") or {}).get("items_over") or []]
    cond = jsonable(db.conditions.find_one({"_id": oid(order["condition_id"])}))
    return _partial(request, "_order.html", o=order, c=cond, over=over)

    return _partial(request, "_ewos.html", job_id=job_id, drafts=_drafts(db, job_id),
                    ewos=jsonable(list(db.ewos.find({"job_id": job_id}).sort("_id", -1))))


@router.get("/ui/jobs/{job_id}/panel/tracking", response_class=HTMLResponse)
def panel_tracking(request: Request, job_id: str, alt: str | None = None):
    db = get_db()
    rows = []
    for o in jsonable(list(db.orders.find({"job_id": job_id, **(scope(job_id, alt) if alt else {})}))):
        state = "sent" if o.get("sent_at") else ("guard tripped" if (o.get("guard") or {}).get("over") else "priced")
        rows.append({"kind": "Order", "id": o["id"], "what": f'{o.get("lf", 0):g} LF', "state": state, "total": o.get("total"),
                     "at": o.get("sent_at") or o.get("created_at")})
    for e in jsonable(list(db.ewos.find({"job_id": job_id}))):
        rows.append({"kind": "EWO", "id": e["id"], "what": e["description"][:70], "state": e["status"], "total": None, "at": e.get("created_at")})
    rows.sort(key=lambda r: r["at"] or "", reverse=True)
    return _partial(request, "_tracking.html", rows=rows)
```
